// FUNGSI PUSH KHUSUS KE TABEL ODP (UPSERT / ANTI DUPLIKAT)
// ==========================================
function insertOdpToSupabase(payloadArray) {
  if (!payloadArray || payloadArray.length === 0) return;
  
  // UBAH PATOKAN: Gunakan kode_odp sebagai patokan agar tidak salah timpa antar stasiun
  const url = `${SUPABASE_URL}/rest/v1/odp?on_conflict=kode_odp`;
  
  const options = {
    method: "post",
    contentType: "application/json",
    headers: {
      "apikey": SUPABASE_KEY,
      "Authorization": "Bearer " + SUPABASE_KEY,
      "Prefer": "resolution=merge-duplicates"
    },
    payload: JSON.stringify(payloadArray),
    muteHttpExceptions: true
  };
  
  try {
    const res = UrlFetchApp.fetch(url, options);
    Logger.log("Response Supabase ODP: " + res.getContentText() || res.getResponseCode());
  } catch (error) {
    Logger.log("Error Supabase ODP: " + error.message);
  }
}

// FUNGSI PERBAIKAN ALAMAT MASSAL KE SUPABASE
// (V2 BULK READ AGAR TIDAK TIMEOUT)
// ==========================================
function perbaikiSemuaAlamatKeSupabase() {
  const stasiunList = Object.keys(STATION_DB_MAP);
  let totalSukses = 0;

  // 1. TARIK SELURUH DATA SUPABASE DI AWAL (KILAT)
  let allSupabaseData = {};
  let start = 0;
  const limit = 1000;
  let hasMore = true;

  Logger.log("Mengunduh data dari Supabase...");
  while (hasMore) {
    let url = SUPABASE_URL + "/rest/v1/data_pelanggan?select=id_pelanggan,alamat&limit=" + limit + "&offset=" + start;
    let options = {
      method: "get",
      headers: {
        "apikey": SUPABASE_KEY,
        "Authorization": "Bearer " + SUPABASE_KEY
      },
      muteHttpExceptions: true
    };
    
    let res = UrlFetchApp.fetch(url, options);
    if (res.getResponseCode() === 200) {
      let chunk = JSON.parse(res.getContentText());
      chunk.forEach(row => {
        if (row.id_pelanggan) {
          allSupabaseData[row.id_pelanggan.toUpperCase()] = row.alamat;
        }
      });
      if (chunk.length < limit) hasMore = false;
      else start += limit;
    } else {
      hasMore = false;
      Logger.log("Gagal mengambil data dari Supabase.");
      return;
    }
  }
  Logger.log("Berhasil mengunduh " + Object.keys(allSupabaseData).length + " data dari Supabase.");

  // 2. BANDINGKAN DENGAN GOOGLE SHEETS
  for (let s = 0; s < stasiunList.length; s++) {
    const stationName = stasiunList[s];
    const stationInfo = STATION_DB_MAP[stationName];
    if (!stationInfo) continue;

    Logger.log("Mengecek stasiun: " + stationName);
    const sheet = SpreadsheetApp.openById(stationInfo.id).getSheetByName(stationInfo.sheetName);
    if (!sheet) continue;

    const data = sheet.getDataRange().getValues();
    let suksesStasiun = 0;

    for (let i = 1; i < data.length; i++) {
      let id = String(data[i][0]).trim().toUpperCase();
      let alamatSheet = String(data[i][3]).trim(); 
      
      // Jika di sheet alamatnya valid
      if (id !== "" && id !== "ID" && alamatSheet !== "" && alamatSheet !== "-") {
        let alamatSupabase = allSupabaseData[id];
        
        // Jika di Supabase alamatnya kosong, null, atau strip "-"
        if (alamatSupabase === undefined || alamatSupabase === null || String(alamatSupabase).trim() === "" || String(alamatSupabase).trim() === "-") {
          
          // Lakukan eksekusi perbaikan (PATCH)
          let urlPatch = SUPABASE_URL + "/rest/v1/data_pelanggan?id_pelanggan=eq." + id;
          let optionsPatch = {
            method: "patch", // Hanya menimpa kolom alamat
            contentType: "application/json",
            headers: {
              "apikey": SUPABASE_KEY,
              "Authorization": "Bearer " + SUPABASE_KEY
            },
            payload: JSON.stringify({ "alamat": alamatSheet }),
            muteHttpExceptions: true
          };
          
          UrlFetchApp.fetch(urlPatch, optionsPatch);
          suksesStasiun++;
          totalSukses++;
        }
      }
    }
    Logger.log("Selesai stasiun " + stationName + ": " + suksesStasiun + " alamat diperbarui.");
  }
  
  Logger.log("Selesai Total! Berhasil memperbaiki " + totalSukses + " alamat di seluruh stasiun.");
}

// MIGRASI ODP HANYA UNTUK 1 STASIUN (MANUAL)
// ==========================================
function migrasiOdpSatuStasiun() {
  // ⬇️ SILAKAN UBAH NAMA DAN ID SPREADSHEET DI BAWAH INI ⬇️
  const TARGET_STASIUN = "Brumbung"; 
  const SPREADSHEET_ID = "1CyHkgCR-6kUO5gxKBQbNwT28PYHTPNDvaSt3F5TXGUs"; 
  // ⬆️ =================================================== ⬆️

  const namaSheetOdp = "ODP"; 
  
  Logger.log("Mulai migrasi ODP khusus stasiun: " + TARGET_STASIUN);
  
  try {
    let ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = ss.getSheetByName(namaSheetOdp);
    
    if (!sheet) {
       Logger.log("=> Sheet '" + namaSheetOdp + "' TIDAK DITEMUKAN di stasiun " + TARGET_STASIUN);
       return; 
    }
    
    let data = sheet.getDataRange().getValues();
    let payloadArray = [];
    let totalBerhasil = 0;
    
    // Asumsi baris ke-1 adalah Header, jadi loop mulai dari index 1
    for (let i = 1; i < data.length; i++) { 
      let row = data[i];
      let label_odp = String(row[0] || "").trim(); // Kolom A
      
      if (label_odp !== "" && label_odp.toLowerCase() !== "label") {
        payloadArray.push({
          label: label_odp,
          latitude: String(row[1] || ""),             // Kolom B
          longitude: String(row[2] || ""),            // Kolom C
          port_terpakai: Number(row[3]) || 0,         // Kolom D
          tahap_pembangunan: String(row[4] || ""),    // Kolom E
          kapasitas: Number(row[5]) || 8,             // Kolom F
          kode_odp: String(row[6] || ""),             // Kolom G
          kode_odc: String(row[7] || ""),             // Kolom H
          stasiun: String(row[8] || TARGET_STASIUN).trim() // Kolom I
        });
      }
      
      // Kirim ke Supabase per 500 data agar tidak timeout/terlalu besar
      if (payloadArray.length === 500) {
         insertOdpToSupabase(payloadArray);
         totalBerhasil += payloadArray.length;
         payloadArray = []; 
      }
    }
    
    // Kirim sisa data yang belum ter-push
    if (payloadArray.length > 0) {
      insertOdpToSupabase(payloadArray);
      totalBerhasil += payloadArray.length;
    }
    
    Logger.log("✅ Sukses migrasi ODP: " + TARGET_STASIUN + ". Total data diproses: " + totalBerhasil);
    
  } catch (e) {
    Logger.log("❌ Error pada stasiun " + TARGET_STASIUN + ": " + e.message);
  }
}
