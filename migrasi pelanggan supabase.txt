function upsertToSupabase(payloadArray) {
  if (!payloadArray || payloadArray.length === 0) return;
  
  const url = `${SUPABASE_URL}/rest/v1/data_pelanggan?on_conflict=id_pelanggan`;
  const options = {
    method: "post",
    contentType: "application/json",
    headers: {
      "apikey": SUPABASE_KEY,
      "Authorization": "Bearer " + SUPABASE_KEY,
      "Prefer": "resolution=merge-duplicates" 
    },
    payload: JSON.stringify(payloadArray),
    muteHttpExceptions: true
  };
  
  try {
    const res = UrlFetchApp.fetch(url, options);
    Logger.log("Response Supabase: " + res.getContentText());
  } catch (error) {
    Logger.log("Error Supabase: " + error.message);
  }
}

// ==========================================
// FUNGSI BANTUAN FORMAT DATA
// ==========================================
function formatTanggalData(val) {
  if (!val) return "";
  if (val instanceof Date) return Utilities.formatDate(val, "Asia/Jakarta", "yyyy-MM-dd HH:mm:ss");
  return String(val);
}

function ekstrakLink(formula, text) {
  if (formula && formula.toString().toUpperCase().startsWith('=HYPERLINK')) {
    let match = formula.match(/=HYPERLINK\(\s*["']([^"']+)["']/i);
    if (match && match[1]) return match[1];
  }
  return String(text || "");
}
