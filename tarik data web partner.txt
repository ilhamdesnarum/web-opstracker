// [+] EKSTRAKSI NAMA SALES DARI PARTNER API
function extractSalesName(customer) {
    if (customer.sales_id && typeof customer.sales_id === 'object' && customer.sales_id.name) {
        return String(customer.sales_id.name).trim();
    }
    if (customer.customer_id && typeof customer.customer_id === 'object' && customer.customer_id.sales_id) {
        var sObj = customer.customer_id.sales_id;
        if (typeof sObj === 'object' && sObj.name) return String(sObj.name).trim();
        if (typeof sObj === 'string' && sObj.trim()) return sObj.trim();
    }
    if (typeof customer.sales_id === 'string' && customer.sales_id.trim()) {
        return customer.sales_id.trim();
    }
    return "Daftar Mandiri";
}

// FUNGSI CEK DATA EXISTING DI SUPABASE
// ==========================================
function checkExistingInSupabase(customerIds, existingMap) {
    if (!customerIds || customerIds.length === 0) return;
    if (typeof SUPABASE_URL === 'undefined' || typeof SUPABASE_KEY === 'undefined') return;

    var idsToQuery = customerIds.filter(function(id) {
        if (!id) return false;
        var sId = id.toString().trim();
        return !existingMap.hasOwnProperty(sId) && !existingMap.hasOwnProperty(sId.toUpperCase());
    });
    if (idsToQuery.length === 0) return;

    var cleanIds = idsToQuery.map(function(id) {
        return id.toString().trim().replace(/['"(),]/g, '');
    }).filter(Boolean);

    if (cleanIds.length === 0) return;

    // Batasi chunk max 50 ID per query URL GET Supabase
    for (var c = 0; c < cleanIds.length; c += 50) {
        var chunk = cleanIds.slice(c, c + 50);
        var url = SUPABASE_URL + "/rest/v1/data_pelanggan?id_pelanggan=in.(" + chunk.join(",") + ")&select=id_pelanggan,status_ikr,status_aktivasi,issue_kendala,latitude,longitude,nama_sales";
        var options = {
            method: "get",
            headers: {
                "apikey": SUPABASE_KEY,
                "Authorization": "Bearer " + SUPABASE_KEY
            },
            muteHttpExceptions: true
        };

        try {
            var res = UrlFetchApp.fetch(url, options);
            if (res.getResponseCode() === 200) {
                var found = JSON.parse(res.getContentText());
                for (var f = 0; f < found.length; f++) {
                    if (found[f] && found[f].id_pelanggan) {
                        var cId = found[f].id_pelanggan.toString().trim();
                        existingMap[cId] = found[f];
                        existingMap[cId.toUpperCase()] = found[f];
                    }
                }
            }
        } catch (err) {
            Logger.log("⚠️ Peringatan cek existing Supabase: " + err.message);
        }
    }
}

function syncPartnerData(statusesToFetch, targetStationName, customSheetInfo, skipSupabase = false, onlyUpdateExisting = true, startPage = 1) {
    let stations = Object.keys(PARTNER_STATION_IDS);

    if (targetStationName) {
        if (Array.isArray(targetStationName)) {
            stations = targetStationName;
        } else if (PARTNER_STATION_IDS[targetStationName]) {
            stations = [targetStationName];
        } else {
            Logger.log("GAGAL: Stasiun '" + targetStationName + "' tidak ditemukan di konfigurasi PARTNER_STATION_IDS.");
            return;
        }
    }


    const options = {
        "method": "get",
        "headers": {
            "Authorization": PARTNER_TOKEN,
            "Cookie": PARTNER_COOKIE
        },
        "muteHttpExceptions": true
    };

    // maxPagesToCheck sekarang diatur langsung di dalam loop (max 5 halaman per status dengan page_size 500)

    var startTime = new Date().getTime();
    var scriptProperties = PropertiesService.getScriptProperties();
    var savedResumeData = scriptProperties.getProperty("SYNC_RESUME_STATE");
    var resumeState = savedResumeData ? JSON.parse(savedResumeData) : null;
    var isTimeout = false;

    let startIdx = 0;
    if (resumeState && resumeState.station) {
        let foundIdx = stations.indexOf(resumeState.station);
        if (foundIdx !== -1) {
            startIdx = foundIdx;
            Logger.log("⏩ Resume aktif: Melompati stasiun yang sudah selesai, mulai dari stasiun: " + resumeState.station);
        }
    }

    for (let sIdx = startIdx; sIdx < stations.length; sIdx++) {
        let stationName = stations[sIdx];
        let partnerId = PARTNER_STATION_IDS[stationName];
        Logger.log("Mulai memproses stasiun: " + stationName + " (ID: " + partnerId + ")");

        try {
            // Jika customSheetInfo diberikan, gunakan itu. Jika tidak, ambil dari STATION_DB_MAP
            const stationInfo = customSheetInfo || (typeof STATION_DB_MAP !== 'undefined' ? STATION_DB_MAP[stationName] : null);
            if (!stationInfo) {
                Logger.log("Bypass: Konfigurasi sheet stasiun " + stationName + " tidak ditemukan.");
                continue;
            }

            var sheet = SpreadsheetApp.openById(stationInfo.id).getSheetByName(stationInfo.sheetName);
            if (!sheet) {
                Logger.log("Bypass: Sheet tidak ditemukan untuk stasiun " + stationName);
                continue;
            }

            // --- LOGIKA MENCARI BARIS TERAKHIR & DATA EXISTING ---
            var aValues = sheet.getRange("A:A").getValues();
            var realLastRow = 0;
            for (var r = aValues.length - 1; r >= 0; r--) {
                if (aValues[r] && aValues[r][0] !== undefined && aValues[r][0].toString().trim() !== "") {
                    realLastRow = r + 1;
                    break;
                }
            }

            var existingIds = {};
            if (realLastRow > 1) {
                var idValues = sheet.getRange(2, 1, realLastRow - 1, 1).getValues();
                for (var r = 0; r < idValues.length; r++) {
                    var id = idValues[r][0].toString().trim();
                    if (id !== "") { existingIds[id] = r + 2; existingIds[id.toUpperCase()] = r + 2; }
                }
            }

            // OPTIMASI BATCH UPDATE: Ambil kolom 9 (Status), 38 (Tgl), dan Sales ke memory
            // Agar tidak timeout gara-gara setValue berulang kali
            var headerRow = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 38)).getValues()[0];
            var idxSalesCol = -1;
            for (var h = 0; h < headerRow.length; h++) {
                var hName = String(headerRow[h] || '').toLowerCase().trim();
                if (hName === 'sales' || hName === 'nama sales' || hName === 'marketing') {
                    idxSalesCol = h;
                    break;
                }
            }
            if (idxSalesCol === -1 && headerRow.length >= 25) {
                idxSalesCol = 24; // Kolom Y (SALES)
            }

            var statusColValues = [];
            var ikrColValues = [];
            var tglColValues = [];
            var latColValues = [];
            var lngColValues = [];
            var odpColValues = [];
            var portOdpColValues = [];
            var salesColValues = [];
            var isColValuesModified = false;
            var isLatLngModified = false;
            var isOdpModified = false;
            var isSalesModified = false;

            if (realLastRow > 0) {
                latColValues = sheet.getRange(1, 6, realLastRow, 1).getValues();
                lngColValues = sheet.getRange(1, 7, realLastRow, 1).getValues();
                ikrColValues = sheet.getRange(1, 8, realLastRow, 1).getValues();
                statusColValues = sheet.getRange(1, 9, realLastRow, 1).getValues();
                odpColValues = sheet.getRange(1, 13, realLastRow, 1).getValues();   // Kolom M (13 - ODP)
                portOdpColValues = sheet.getRange(1, 14, realLastRow, 1).getValues(); // Kolom N (14 - Port ODP)
                tglColValues = sheet.getRange(1, 38, realLastRow, 1).getValues();
                if (idxSalesCol !== -1) {
                    salesColValues = sheet.getRange(1, idxSalesCol + 1, realLastRow, 1).getValues();
                }
            }

            var totalUpdateTanggal = 0;
            var totalBaru = 0;

            var existingSupabaseMap = {};
            var supabasePayloadUpdateWithStatus = [];
            var supabasePayloadExistingNoStatus = [];
            var supabasePayloadNew = [];
            var newRowsSheet = [];

            // Variabel startTime, isTimeout, resumeState sudah diinisialisasi sebelum loop stations

            for (var s = 0; s < statusesToFetch.length; s++) {
                var currentStatus = statusesToFetch[s];
                var page = startPage;

                if (resumeState && resumeState.station === stationName) {
                    if (resumeState.status !== currentStatus && !resumeState.passed) {
                        Logger.log("⏩ Melewati status '" + currentStatus + "' (sudah selesai diproses di sesi sebelumnya)");
                        continue; // Lewati status ini karena sudah selesai di sesi sebelumnya
                    } else if (resumeState.status === currentStatus && !resumeState.passed) {
                        page = resumeState.page;
                        resumeState.passed = true; // Tandai titik temu resume
                        Logger.log("🔄 MELANJUTKAN OTOMATIS STATUS '" + currentStatus + "' DARI HALAMAN " + page);
                    }
                }

                var hasMoreData = true;
                var pageSize = 10; // Ukuran baku server Partner terbaru (1 Halaman = 10 Data)
                var maxPages = 150; // 150 x 30 = 4500 pelanggan (mencakup seluruh stasiun besar seperti Kradenan 2200+ data)

                while (hasMoreData && page <= maxPages) {
                    // Batas waktu aman (3.0 menit) agar data tetap tersimpan sebelum Google Apps Script timeout 6 menit
                    if (new Date().getTime() - startTime > 180000) {
                        Logger.log("⚠️ Waktu eksekusi mencapai 3.0 menit. Menyimpan progress (Halaman " + page + ") untuk dilanjutkan nanti...");
                        scriptProperties.setProperty("SYNC_RESUME_STATE", JSON.stringify({
                            station: stationName,
                            status: currentStatus,
                            page: page,
                            passed: false,
                            args: {
                                statusesToFetch: statusesToFetch,
                                targetStationName: targetStationName,
                                customSheetInfo: customSheetInfo,
                                skipSupabase: skipSupabase,
                                onlyUpdateExisting: onlyUpdateExisting
                            }
                        }));
                        hasMoreData = false;
                        isTimeout = true;
                        break;
                    }

                    var urlList = "";
                    if (currentStatus === "new") {
                        urlList = "https://api-mitra.starliteindonesia.com/mitra/customer/new?page=" + page + "&page_size=" + pageSize + "&sort_order=DESC&sales_partner_id=" + partnerId;
                    } else if (currentStatus.startsWith("new-")) {
                        var ikrStatus = currentStatus.replace("new-", "");
                        urlList = "https://api-mitra.starliteindonesia.com/mitra/customer/new?page=" + page + "&page_size=" + pageSize + "&sort_order=DESC&ikr_status=" + ikrStatus + "&sales_partner_id=" + partnerId;
                    } else if (currentStatus === "active") {
                        urlList = "https://api-mitra.starliteindonesia.com/mitra/customer/active?page=" + page + "&page_size=" + pageSize + "&sort_order=DESC&sales_partner_id=" + partnerId;
                    } else {
                        var apiPath = (currentStatus === "dismantle" || currentStatus === "dismantled" || currentStatus === "ready-to-dismantle") ? "suspend" : currentStatus;
                        var apiStatusParam = (currentStatus === "dismantle") ? "dismantled" : currentStatus;
                        urlList = "https://api-mitra.starliteindonesia.com/mitra/customer/" + apiPath + "?page=" + page + "&page_size=" + pageSize + "&sort_order=DESC&status=" + apiStatusParam + "&sales_partner_id=" + partnerId;
                    }

                    var responseList = UrlFetchApp.fetch(urlList, options);
                    if (responseList.getResponseCode() !== 200) {
                        Logger.log("Gagal akses API List " + currentStatus + " di " + stationName + " (Hal " + page + "). Response: " + responseList.getResponseCode());
                        break;
                    }

                    var jsonList = JSON.parse(responseList.getContentText());
                    if (jsonList && jsonList.data && jsonList.data.length > 0) {
                        var customers = jsonList.data;
                        Logger.log("-> " + stationName + " [" + currentStatus + "] Hal " + page + ": memproses " + customers.length + " data...");

                        var isTarikPelangganBaru = currentStatus.startsWith("new") || currentStatus === "new";

                        // Cek apakah data pelanggan di halaman ini sudah ada di Supabase
                        if (!skipSupabase) {
                            var pageCustomerIds = [];
                            for (var pIdx = 0; pIdx < customers.length; pIdx++) {
                                var custObj = customers[pIdx];
                                if (!custObj) continue;
                                var rawPId = custObj.customer_id && typeof custObj.customer_id === 'string' ? custObj.customer_id : (custObj.customer_id ? custObj.customer_id.customer_id : "");
                                var pId = String(rawPId || "").trim();
                                if (pId) {
                                    pageCustomerIds.push(pId);
                                    pageCustomerIds.push(pId.toUpperCase());
                                }
                            }
                            if (pageCustomerIds.length > 0) {
                                checkExistingInSupabase(pageCustomerIds, existingSupabaseMap);
                            }
                        }

                        for (var i = 0; i < customers.length; i++) {
                            var customer = customers[i];
                            if (!customer) continue;

                            var rawCustId = customer.customer_id && typeof customer.customer_id === 'string' ? customer.customer_id : (customer.customer_id ? customer.customer_id.customer_id : "");
                            var idPelanggan = String(rawCustId || "").trim();
                            if (!idPelanggan) continue;
                            var idPelangganUpper = idPelanggan.toUpperCase();

                            // --- PENGECEKAN DATA EXISTING (DEKLARASI DI AWAL) ---
                            var existsInSupabase = (existingSupabaseMap.hasOwnProperty(idPelanggan) && existingSupabaseMap[idPelanggan] !== false) ||
                                                   (existingSupabaseMap.hasOwnProperty(idPelangganUpper) && existingSupabaseMap[idPelangganUpper] !== false);
                            var existsInSheet = existingIds.hasOwnProperty(idPelanggan) || existingIds.hasOwnProperty(idPelangganUpper);
                            var alreadyExists = existsInSupabase || existsInSheet;
                            var targetRow = existsInSheet ? (existingIds[idPelanggan] || existingIds[idPelangganUpper]) : null; // 1-indexed
                            var sbData = existingSupabaseMap[idPelanggan] || existingSupabaseMap[idPelangganUpper];

                            var nama = customer.name || (customer.customer_id ? customer.customer_id.name : "");
                            var telepon = customer.phone_number || (customer.customer_id ? customer.customer_id.phone_number : "");
                            var alamat = customer.address || (customer.customer_id ? customer.customer_id.address : "");
                            var patokan = customer.notes || "";
                            var namaSales = extractSalesName(customer);

                            // --- EKSTRAKSI LATITUDE & LONGITUDE LENGKAP & DEFENSIVE ---
                            var lat = "";
                            var lng = "";

                            // 1. Cek di customer_id jika berupa objek (prioritas utama endpoint /new)
                            if (customer.customer_id && typeof customer.customer_id === 'object') {
                                if (customer.customer_id.latitude !== undefined && customer.customer_id.latitude !== null && customer.customer_id.latitude !== "") {
                                    lat = String(customer.customer_id.latitude).trim().replace(",", ".");
                                }
                                if (customer.customer_id.longitude !== undefined && customer.customer_id.longitude !== null && customer.customer_id.longitude !== "") {
                                    lng = String(customer.customer_id.longitude).trim().replace(",", ".");
                                }
                                if (!lat && customer.customer_id.lat) lat = String(customer.customer_id.lat).trim().replace(",", ".");
                                if (!lng && (customer.customer_id.lng || customer.customer_id.long)) lng = String(customer.customer_id.lng || customer.customer_id.long).trim().replace(",", ".");
                            }

                            // 2. Cek di root object
                            if (!lat && customer.latitude !== undefined && customer.latitude !== null && customer.latitude !== "") {
                                lat = String(customer.latitude).trim().replace(",", ".");
                            }
                            if (!lng && customer.longitude !== undefined && customer.longitude !== null && customer.longitude !== "") {
                                lng = String(customer.longitude).trim().replace(",", ".");
                            }
                            if (!lat && customer.lat !== undefined && customer.lat !== null && customer.lat !== "") {
                                lat = String(customer.lat).trim().replace(",", ".");
                            }
                            if (!lng && (customer.lng !== undefined || customer.long !== undefined)) {
                                lng = String(customer.lng || customer.long).trim().replace(",", ".");
                            }

                            // 3. Cek di installation_info_id
                            if ((!lat || !lng) && customer.installation_info_id) {
                                var instList = Array.isArray(customer.installation_info_id) ? customer.installation_info_id : [customer.installation_info_id];
                                for (var ii = 0; ii < instList.length; ii++) {
                                    var instItem = instList[ii];
                                    if (!instItem) continue;
                                    if (!lat && (instItem.latitude || instItem.lat)) {
                                        lat = String(instItem.latitude || instItem.lat).trim().replace(",", ".");
                                    }
                                    if (!lng && (instItem.longitude || instItem.lng || instItem.long)) {
                                        lng = String(instItem.longitude || instItem.lng || instItem.long).trim().replace(",", ".");
                                    }
                                    if (lat && lng) break;
                                }
                            }

                            // Cek di sales_visit_id jika ada
                            if ((!lat || !lng) && customer.sales_visit_id && typeof customer.sales_visit_id === 'object') {
                                if (!lat && (customer.sales_visit_id.latitude || customer.sales_visit_id.lat)) {
                                    lat = String(customer.sales_visit_id.latitude || customer.sales_visit_id.lat).trim().replace(",", ".");
                                }
                                if (!lng && (customer.sales_visit_id.longitude || customer.sales_visit_id.lng || customer.sales_visit_id.long)) {
                                    lng = String(customer.sales_visit_id.longitude || customer.sales_visit_id.lng || customer.sales_visit_id.long).trim().replace(",", ".");
                                }
                            }

                            // Fallback ke koordinat yang sudah ada di Supabase jika API Partner tidak ada koordinat
                            if ((!lat || !lng) && sbData && typeof sbData === 'object') {
                                if (!lat && sbData.latitude) lat = String(sbData.latitude).trim().replace("'", "").replace(",", ".");
                                if (!lng && sbData.longitude) lng = String(sbData.longitude).trim().replace("'", "").replace(",", ".");
                            }

                            // Fallback jika API Partner tidak ada koordinat tapi di Google Sheet sudah terisi
                            if ((!lat || !lng) && targetRow && targetRow <= realLastRow) {
                                var sheetLatVal = latColValues[targetRow - 1] ? String(latColValues[targetRow - 1][0] || '').trim().replace("'", "").replace(",", ".") : "";
                                var sheetLngVal = lngColValues[targetRow - 1] ? String(lngColValues[targetRow - 1][0] || '').trim().replace("'", "").replace(",", ".") : "";
                                if (!lat && sheetLatVal) lat = sheetLatVal;
                                if (!lng && sheetLngVal) lng = sheetLngVal;
                            }

                            // Sedot ODP & Port ODP dari Partner API (Mendukung struktur installation_info_id & fallback)
                            var odp = "";
                            var portOdp = "";

                            if (customer.installation_info_id) {
                                var instArr = Array.isArray(customer.installation_info_id) ? customer.installation_info_id : [customer.installation_info_id];
                                if (instArr.length > 0 && instArr[0]) {
                                    var inst = instArr[0];
                                    portOdp = (inst.port_number !== undefined && inst.port_number !== null) ? String(inst.port_number) : (inst.port ? String(inst.port) : "");
                                    if (inst.device_id) {
                                        odp = inst.device_id.code || inst.device_id.name || "";
                                    }
                                }
                            }

                            // Fallback jika ada properti tingkat atas
                            if (!odp) odp = customer.odp || customer.odp_name || customer.odp_code || customer.fat || customer.fat_code || customer.fat_name || (customer.customer_id && typeof customer.customer_id === 'object' ? (customer.customer_id.odp || customer.customer_id.fat_code || customer.customer_id.fat_name) : "") || "";
                            if (!portOdp) portOdp = customer.port_odp || customer.port || customer.fat_port || customer.odp_port || (customer.customer_id && typeof customer.customer_id === 'object' ? (customer.customer_id.port_odp || customer.customer_id.port) : "") || "";

                            var tglRegistrasi = "";
                            if (customer.registration_date) tglRegistrasi = formatKeWIB(customer.registration_date);
                            else if (customer.visit_date) tglRegistrasi = formatKeWIB(customer.visit_date);
                            else if (customer.sales_visit_id && customer.sales_visit_id.visit_date) tglRegistrasi = formatKeWIB(customer.sales_visit_id.visit_date);
                            else if (customer.customer_id && typeof customer.customer_id === 'object' && customer.customer_id.registration_date) tglRegistrasi = formatKeWIB(customer.customer_id.registration_date);
                            else if (customer.created_at) tglRegistrasi = formatKeWIB(customer.created_at);

                            // --- EKSTRAK TANGGAL BERAKHIR & TELAT BAYAR ---
                            var tanggalBerakhir = "";
                            var telatBayarHari = (customer.count_late_payment_days !== undefined && customer.count_late_payment_days !== null && customer.count_late_payment_days !== "") ? Number(customer.count_late_payment_days) : null;

                            if (customer.user_ppoe_id && customer.user_ppoe_id.yinet_info) {
                                try {
                                    var yinet = typeof customer.user_ppoe_id.yinet_info === 'string' ? JSON.parse(customer.user_ppoe_id.yinet_info) : customer.user_ppoe_id.yinet_info;
                                    if (yinet && yinet.expired_date) tanggalBerakhir = formatKeWIB(yinet.expired_date);
                                } catch(e) {}
                            }
                            if (!tanggalBerakhir && customer.ont_id && customer.ont_id.length > 0 && customer.ont_id[0].payload) {
                                try {
                                    var payloadOnt = typeof customer.ont_id[0].payload === 'string' ? JSON.parse(customer.ont_id[0].payload) : customer.ont_id[0].payload;
                                    if (payloadOnt && payloadOnt.expired_date) tanggalBerakhir = formatKeWIB(payloadOnt.expired_date);
                                } catch(e) {}
                            }

                            var tStatus = "Sudah";
                            var tIkr = "Sudah";
                            if (currentStatus === "suspend") tStatus = "Suspend";
                            else if (currentStatus === "dismantle") tStatus = "Dismantled";
                            else if (currentStatus === "ready-to-dismantle") tStatus = "Ready To Dismantle";
                            else if (currentStatus.startsWith("new")) {
                                tStatus = "Belum";
                                tIkr = "Belum";
                            }

                            // 1. Cek nilai status eksisting di Google Sheet saat ini
                            var isCurrentlyKendalaInSheet = false;
                            if (targetRow && targetRow <= realLastRow) {
                                var currIkrVal = String(ikrColValues[targetRow - 1][0] || '').trim().toUpperCase();
                                var currStatVal = String(statusColValues[targetRow - 1][0] || '').trim().toUpperCase();
                                if (currIkrVal.includes("KENDALA") || currStatVal.includes("KENDALA") || currIkrVal.includes("CANCEL") || currStatVal.includes("CANCEL")) {
                                    isCurrentlyKendalaInSheet = true;
                                }
                            }

                            // 2. Cek nilai status eksisting di Supabase saat ini
                            var isCurrentlyKendalaInSupabase = false;
                            if (sbData && typeof sbData === 'object') {
                                var sbIkr = String(sbData.status_ikr || '').trim().toUpperCase();
                                var sbStat = String(sbData.status_aktivasi || '').trim().toUpperCase();
                                if (sbIkr.includes("KENDALA") || sbStat.includes("KENDALA") || (sbData.issue_kendala && String(sbData.issue_kendala).trim() !== '')) {
                                    isCurrentlyKendalaInSupabase = true;
                                }
                            }

                            var isProtectedKendala = isCurrentlyKendalaInSheet || isCurrentlyKendalaInSupabase;

                            var cleanLat = lat ? String(lat).replace(/^'+/, "").trim().replace(",", ".") : "";
                            var cleanLng = lng ? String(lng).replace(/^'+/, "").trim().replace(",", ".") : "";

                            if (alreadyExists) {
                                // DATA LAMA / SUDAH ADA DI DATABASE (PELANGGAN LAMA)
                                // [PROTEKSI MUTLAK]:
                                // 1. JANGAN PERNAH perbarui Latitude & Longitude (sudah presisi saat aktivasi)
                                // 2. JANGAN PERNAH timpa Nama Sales (pelanggan lama sudah disinkronkan manual)

                                // 1. UPDATE GOOGLE SHEET
                                if (targetRow && targetRow <= realLastRow) {
                                    // [!] PROTEKSI MUTLAK:
                                    // - JANGAN timpa jika isTarikPelangganBaru = true
                                    // - JANGAN timpa jika data saat ini sudah berstatus KENDALA di Sheet/Supabase
                                    if (!isTarikPelangganBaru && !isProtectedKendala) {
                                        ikrColValues[targetRow - 1][0] = tIkr;
                                        statusColValues[targetRow - 1][0] = tStatus;
                                        isColValuesModified = true;
                                    }
                                } else if (targetRow) {
                                    var newRowIndex = targetRow - realLastRow - 2;
                                    if (newRowIndex >= 0 && newRowIndex < newRowsSheet.length) {
                                        if (!isTarikPelangganBaru && !isProtectedKendala) {
                                            newRowsSheet[newRowIndex][7] = tIkr;
                                            newRowsSheet[newRowIndex][8] = tStatus;
                                        }
                                    }
                                }

                                // 2. UPDATE KE SUPABASE
                                if (!isTarikPelangganBaru && !isProtectedKendala) {
                                    // Normal status update (Active, Suspend, Dismantle) - TANPA koordinat & TANPA nama_sales
                                    var updatePayload = {
                                        id_pelanggan: idPelanggan,
                                        status_ikr: tIkr,
                                        status_aktivasi: tStatus,
                                        tanggal_berakhir: tanggalBerakhir || null,
                                        telat_bayar_hari: telatBayarHari
                                    };
                                    supabasePayloadUpdateWithStatus.push(updatePayload);
                                    totalUpdateTanggal++;
                                } else {
                                    // [!] KHUSUS TARIK PELANGGAN BARU ATAU STATUS KENDALA:
                                    // Update data kontak/alamat saja - TANPA menimpa koordinat, nama_sales, status_ikr, & status_aktivasi
                                    var existingUpdatePayload = {
                                        id_pelanggan: idPelanggan,
                                        nama_pelanggan: nama,
                                        nomor_hp: telepon,
                                        alamat: alamat,
                                        catatan: patokan,
                                        tanggal_registrasi: tglRegistrasi,
                                        tanggal_berakhir: tanggalBerakhir || null,
                                        telat_bayar_hari: telatBayarHari,
                                        stasiun: stationName,
                                        odp: odp,
                                        port_odp: portOdp
                                    };
                                    supabasePayloadExistingNoStatus.push(existingUpdatePayload);
                                    totalUpdateTanggal++;
                                }
                            } else if (!onlyUpdateExisting) {
                                // DATA BENAR-BENAR BARU
                                totalBaru++;
                                existingIds[idPelanggan] = realLastRow + totalBaru + 1;
                                existingIds[idPelangganUpper] = realLastRow + totalBaru + 1;
                                existingSupabaseMap[idPelanggan] = true;
                                existingSupabaseMap[idPelangganUpper] = true;

                                var totalCols = Math.max(38, idxSalesCol !== -1 ? idxSalesCol + 1 : 38);
                                var newRow = new Array(totalCols).fill("");
                                newRow[0] = idPelanggan;
                                newRow[1] = nama;
                                newRow[2] = telepon;
                                newRow[3] = alamat;
                                newRow[4] = patokan;
                                newRow[5] = cleanLat ? "'" + cleanLat : ""; // Format string teks biasa
                                newRow[6] = cleanLng ? "'" + cleanLng : ""; // Format string teks biasa
                                newRow[7] = tIkr;
                                newRow[8] = tStatus;
                                if (idxSalesCol !== -1) newRow[idxSalesCol] = namaSales;
                                newRow[12] = odp;     // Kolom M (13 - ODP)
                                newRow[13] = portOdp; // Kolom N (14 - Port ODP)
                                newRow[32] = stationName;
                                newRow[37] = tglRegistrasi;
                                newRowsSheet.push(newRow);

                                supabasePayloadNew.push({
                                    id_pelanggan: idPelanggan,
                                    nama_pelanggan: nama,
                                    nomor_hp: telepon,
                                    alamat: alamat,
                                    catatan: patokan,
                                    nama_sales: namaSales,
                                    latitude: cleanLat,
                                    longitude: cleanLng,
                                    status_ikr: tIkr,
                                    status_aktivasi: tStatus,
                                    tanggal_registrasi: tglRegistrasi,
                                    tanggal_berakhir: tanggalBerakhir || null,
                                    telat_bayar_hari: telatBayarHari,
                                    stasiun: stationName,
                                    odp: odp,
                                    port_odp: portOdp
                                });
                            }
                        }
                        if (customers.length < pageSize) {
                            hasMoreData = false;
                        } else {
                            page++;
                            Utilities.sleep(100); // Jeda singkat 0.1s
                        }
                    } else {
                        hasMoreData = false;
                    }
                }
                if (isTimeout) break; // Berhenti memproses status lainnya jika sudah timeout
            }


            // --- PUSH KE GOOGLE SHEET ---
            if (isColValuesModified && realLastRow > 0) {
                sheet.getRange(1, 8, realLastRow, 1).setValues(ikrColValues);
                sheet.getRange(1, 9, realLastRow, 1).setValues(statusColValues);
            }

            if (isLatLngModified && realLastRow > 0) {
                // [PENTING] Set format teks biasa (@) untuk kolom Latitude (6) dan Longitude (7)
                // agar Google Sheets TIDAK mengubah koordinat menjadi format angka ribuan
                var updateRangeLat = sheet.getRange(1, 6, realLastRow, 1);
                var updateRangeLng = sheet.getRange(1, 7, realLastRow, 1);
                updateRangeLat.setNumberFormat("@");
                updateRangeLng.setNumberFormat("@");
                updateRangeLat.setValues(latColValues);
                updateRangeLng.setValues(lngColValues);
                updateRangeLat.setNumberFormat("@");
                updateRangeLng.setNumberFormat("@");
            }

            if (isSalesModified && realLastRow > 0 && idxSalesCol !== -1) {
                sheet.getRange(1, idxSalesCol + 1, realLastRow, 1).setValues(salesColValues);
            }

            if (isOdpModified && realLastRow > 0) {
                sheet.getRange(1, 13, realLastRow, 1).setValues(odpColValues);  // Kolom M (ODP)
                sheet.getRange(1, 14, realLastRow, 1).setValues(portOdpColValues); // Kolom N (Port ODP)
            }

            if (newRowsSheet.length > 0) {
                // [PENTING] Format sel kolom Latitude dan Longitude sebagai Plain Text (@) sebelum dan sesudah penulisan
                var newRangeLat = sheet.getRange(realLastRow + 1, 6, newRowsSheet.length, 1);
                var newRangeLng = sheet.getRange(realLastRow + 1, 7, newRowsSheet.length, 1);
                newRangeLat.setNumberFormat("@");
                newRangeLng.setNumberFormat("@");
                sheet.getRange(realLastRow + 1, 1, newRowsSheet.length, newRowsSheet[0].length).setValues(newRowsSheet);
                newRangeLat.setNumberFormat("@");
                newRangeLng.setNumberFormat("@");
            }

            // --- PUSH KE SUPABASE (Memanggil Fungsi upsertToSupabase yg ada di file lain) ---
            if (!skipSupabase) {
                // 1. Pelanggan existing dengan update status (Active, Suspend, Dismantle)
                if (supabasePayloadUpdateWithStatus.length > 0) {
                    var uniqueMapUpdateStatus = {};
                    for (var k = 0; k < supabasePayloadUpdateWithStatus.length; k++) {
                        uniqueMapUpdateStatus[supabasePayloadUpdateWithStatus[k].id_pelanggan] = supabasePayloadUpdateWithStatus[k];
                    }
                    if (typeof upsertToSupabase === "function") upsertToSupabase(Object.values(uniqueMapUpdateStatus));
                }

                // 2. Pelanggan existing saat tarik baru (TANPA menimpa status_ikr & status_aktivasi)
                if (supabasePayloadExistingNoStatus.length > 0) {
                    var uniqueMapExistingNoStatus = {};
                    for (var k = 0; k < supabasePayloadExistingNoStatus.length; k++) {
                        uniqueMapExistingNoStatus[supabasePayloadExistingNoStatus[k].id_pelanggan] = supabasePayloadExistingNoStatus[k];
                    }
                    if (typeof upsertToSupabase === "function") upsertToSupabase(Object.values(uniqueMapExistingNoStatus));
                }

                // 3. Pelanggan baru murni (belum ada di database)
                if (supabasePayloadNew.length > 0) {
                    var uniqueMapNew = {};
                    for (var k = 0; k < supabasePayloadNew.length; k++) {
                        uniqueMapNew[supabasePayloadNew[k].id_pelanggan] = supabasePayloadNew[k];
                    }
                    if (typeof upsertToSupabase === "function") upsertToSupabase(Object.values(uniqueMapNew));
                }
            } else {
                Logger.log("Bypass: Sinkronisasi ke Supabase dilewati (skipSupabase = true).");
            }

            Logger.log("Stasiun " + stationName + " selesai. Baru: " + totalBaru + ", Update: " + totalUpdateTanggal);

        } catch (e) {
            Logger.log("Error di stasiun " + stationName + ": " + e.message);
        }

        if (isTimeout) {
            Logger.log("⚠️ Berhenti sementara karena batas waktu. Auto-resume akan menjadwalkan kelanjutannya...");
            createAutoResumeTrigger();
            return;
        }

        // Jeda antar stasiun
        Utilities.sleep(1500);
    }

    scriptProperties.deleteProperty("SYNC_RESUME_STATE");
    Logger.log("✅ Sinkronisasi Partner Semua Stasiun Selesai!");
}

// FUNGSI BANTUAN UNTUK PUSH DATA KE SUPABASE
// ==========================================
function upsertToSupabase(payloadArray) {
    if (!payloadArray || payloadArray.length === 0) return;

    if (typeof SUPABASE_URL === 'undefined' || typeof SUPABASE_KEY === 'undefined') {
        Logger.log("⚠️ Supabase Credentials (SUPABASE_URL / SUPABASE_KEY) tidak ditemukan di lingkup global. Lewati sync Supabase.");
        return;
    }

    var nowIso = new Date().toISOString();
    payloadArray.forEach(function(item) {
        // Hanya sertakan created_at untuk data yang memang memilikinya agar tidak menimpa timestamp data lama
        if (!item.created_at) delete item.created_at;
        item.updated_at = nowIso;
    });

    var url = SUPABASE_URL + "/rest/v1/data_pelanggan?on_conflict=id_pelanggan";

    var options = {
        method: "post",
        headers: {
            "apikey": SUPABASE_KEY,
            "Authorization": "Bearer " + SUPABASE_KEY,
            "Content-Type": "application/json",
            "Prefer": "resolution=merge-duplicates" // INI KUNCI UPSERT DI SUPABASE
        },
        payload: JSON.stringify(payloadArray),
        muteHttpExceptions: true
    };

    try {
        var response = UrlFetchApp.fetch(url, options);
        if (response.getResponseCode() >= 400) {
            Logger.log("❌ Gagal Upsert ke Supabase: " + response.getContentText());
        } else {
            Logger.log("✅ Sukses Upsert " + payloadArray.length + " baris ke Supabase.");
        }
    } catch (e) {
        Logger.log("❌ Error koneksi Supabase: " + e.message);
    }
}

// FUNGSI AUTO-RESUME TRIGGER
// ==========================================
function createAutoResumeTrigger() {
    // Hapus trigger lama agar tidak menumpuk
    var triggers = ScriptApp.getProjectTriggers();
    for (var i = 0; i < triggers.length; i++) {
        if (triggers[i].getHandlerFunction() === "autoResumeTrigger") {
            ScriptApp.deleteTrigger(triggers[i]);
        }
    }
    // Buat trigger baru untuk berjalan 1 menit dari sekarang
    ScriptApp.newTrigger("autoResumeTrigger")
        .timeBased()
        .after(60000)
        .create();
    Logger.log("⏱️ Auto-Resume telah dijadwalkan. Script akan otomatis berjalan sendiri dalam 1 menit.");
}

function autoResumeTrigger() {
    Logger.log("▶️ Memulai Auto-Resume...");
    // Bersihkan trigger setelah dijalankan agar tidak berulang
    var triggers = ScriptApp.getProjectTriggers();
    for (var i = 0; i < triggers.length; i++) {
        if (triggers[i].getHandlerFunction() === "autoResumeTrigger") {
            ScriptApp.deleteTrigger(triggers[i]);
        }
    }

    var scriptProperties = PropertiesService.getScriptProperties();
    var savedResumeData = scriptProperties.getProperty("SYNC_RESUME_STATE");
    if (savedResumeData) {
        var state = JSON.parse(savedResumeData);
        if (state.args) {
            syncPartnerData(
                state.args.statusesToFetch, 
                state.args.targetStationName, 
                state.args.customSheetInfo, 
                state.args.skipSupabase, 
                state.args.onlyUpdateExisting, 
                1
            );
        }
    } else {
        Logger.log("Tidak ada state resume yang ditemukan.");
    }
}
